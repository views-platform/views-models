# algorithms — InterpolatedQuantileConflictology, KDEConflictology and

# this arm's `algorithm` names something the registry cannot build.

# views-pipeline-core>=3.0.0,<4.0.0 — a floor BELOW the 3.2.0 that can find
# configs/config_maturity.py. An env holding exactly 3.0.0 satisfies that range, so the
# launcher upgrades nothing and the run dies on a missing config_deployment.py (#594).
views-pipeline-core>=3.2.0,<4.0.0

# dependency with no extra marker, which makes views-frames 3.x unreachable for anything
# sharing this environment. 1.17.0 widened it to `<4` (views-datafactory#527).
views-datafactory>=1.17.0,<2.0.0
